    <script>
        // Cập nhật năm tự động ở footer
        document.getElementById('year').innerText = new Date().getFullYear();

        // 1. CHỨC NĂNG THEME TOGGLE (LIGHT / DARK MODE)
        const themeIcon = document.getElementById('theme-icon');
        const htmlElement = document.documentElement;

        // Khởi tạo theme từ localStorage hoặc mặc định
        const savedTheme = localStorage.getItem('bio_theme') || 'light';
        if (savedTheme === 'dark') {
            htmlElement.classList.add('dark');
            themeIcon.className = 'fa-solid fa-sun text-lg text-amber-400';
        } else {
            htmlElement.classList.remove('dark');
            themeIcon.className = 'fa-solid fa-moon text-lg text-sky-500';
        }

        function toggleTheme() {
            if (htmlElement.classList.contains('dark')) {
                htmlElement.classList.remove('dark');
                themeIcon.className = 'fa-solid fa-moon text-lg text-sky-500';
                localStorage.setItem('bio_theme', 'light');
            } else {
                htmlElement.classList.add('dark');
                themeIcon.className = 'fa-solid fa-sun text-lg text-amber-400';
                localStorage.setItem('bio_theme', 'dark');
            }
        }

        // 2. CHỨC NĂNG MODAL & TẠO MA QR CODE DỘNG
        const modal = document.getElementById('share-modal');
        const modalContent = document.getElementById('modal-content');
        const shareUrlInput = document.getElementById('share-url-input');
        let qrcodeObj = null;

        function openShareModal() {
            // Lấy URL trang web hiện tại
            const currentUrl = window.location.href;
            shareUrlInput.value = currentUrl;

            // Render Mã QR
            const qrContainer = document.getElementById('qrcode');
            qrContainer.innerHTML = ''; // Clear cũ
            
            qrcodeObj = new QRCode(qrContainer, {
                text: currentUrl,
                width: 140,
                height: 140,
                colorDark: "#0284C7",
                colorLight: "#FFFFFF",
                correctLevel: QRCode.CorrectLevel.H
            });

            // Hiển thị modal với hiệu ứng zoom-in
            modal.classList.remove('hidden');
            setTimeout(() => {
                modal.classList.remove('opacity-0');
                modalContent.classList.remove('scale-95');
                modalContent.classList.add('scale-100');
            }, 10);
        }

        function closeShareModal() {
            modal.classList.add('opacity-0');
            modalContent.classList.remove('scale-100');
            modalContent.classList.add('scale-95');
            setTimeout(() => {
                modal.classList.add('hidden');
            }, 300);
        }

        // Đóng modal khi bấm ra ngoài khung
        modal.addEventListener('click', function(e) {
            if (e.target === modal) {
                closeShareModal();
            }
        });

        // 3. CHỨC NĂNG SAO CHÉP LINK CLIPBOARD
        function copyToClipboard() {
            const urlText = shareUrlInput.value;
            
            // Lựa chọn sao chép an toàn cho mọi trình duyệt và iframe
            shareUrlInput.select();
            shareUrlInput.setSelectionRange(0, 99999);

            try {
                document.execCommand('copy');
                showToast("Đã sao chép liên kết vào bộ nhớ tạm!");
            } catch (err) {
                navigator.clipboard.writeText(urlText).then(() => {
                    showToast("Đã sao chép liên kết vào bộ nhớ tạm!");
                }).catch(e => showToast("Không thể sao chép, vui lòng tự copy!"));
            }
        }

        function showToast(msg) {
            const toast = document.getElementById('modal-toast');
            const toastText = document.getElementById('toast-text');
            const copyBtnText = document.getElementById('copy-btn-text');

            toastText.innerText = msg;
            copyBtnText.innerText = 'Đã chép!';
            toast.classList.remove('opacity-0');

            setTimeout(() => {
                copyBtnText.innerText = 'Sao chép';
                toast.classList.add('opacity-0');
            }, 2500);
        }

        // 4. CHỨC NĂNG TẢI MÃ QR VỀ MÁY
        function downloadQRCode() {
            const qrImg = document.querySelector('#qrcode img');
            const qrCanvas = document.querySelector('#qrcode canvas');

            let imageSrc = '';
            if (qrImg && qrImg.src) {
                imageSrc = qrImg.src;
            } else if (qrCanvas) {
                imageSrc = qrCanvas.toDataURL('image/png');
            }

            if (imageSrc) {
                const link = document.createElement('a');
                link.href = imageSrc;
                link.download = 'bio-qrcode.png';
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
            } else {
                alert('Mã QR đang được khởi tạo, vui lòng thử lại sau giây lát!');
            }
        }
    </script>
