    <!-- JAVASCRIPT CONTROLLER LOGIC -->
    <script>
        // Set dynamic copyright year
        document.getElementById('year').innerText = new Date().getFullYear();

        // 1. LIGHT / DARK MODE CONTROLLER
        const themeIcon = document.getElementById('theme-icon');
        const htmlElement = document.documentElement;

        const savedTheme = localStorage.getItem('bio_theme') || 'light';
        if (savedTheme === 'dark') {
            htmlElement.classList.add('dark');
            themeIcon.className = 'fa-solid fa-sun text-base text-amber-400';
        } else {
            htmlElement.classList.remove('dark');
            themeIcon.className = 'fa-solid fa-moon text-base text-sky-500';
        }

        function toggleTheme() {
            if (htmlElement.classList.contains('dark')) {
                htmlElement.classList.remove('dark');
                themeIcon.className = 'fa-solid fa-moon text-base text-sky-500';
                localStorage.setItem('bio_theme', 'light');
            } else {
                htmlElement.classList.add('dark');
                themeIcon.className = 'fa-solid fa-sun text-base text-amber-400';
                localStorage.setItem('bio_theme', 'dark');
            }
        }

        // 2. AVATAR PREVIEW UPLOAD
        function previewAvatar(event) {
            const file = event.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    document.getElementById('user-avatar').src = e.target.result;
                    showToast('Đã cập nhật ảnh đại diện tạm thời!');
                };
                reader.readAsDataURL(file);
            }
        }

        // 3. SHARE OR COPY LINK (WEB SHARE API WITH FALLBACK)
        function shareOrCopyLink() {
            const currentUrl = window.location.href;
            if (navigator.share) {
                navigator.share({
                    title: 'Bio Link Cá Nhân',
                    text: 'Trang thông tin liên hệ của Nhật Huy',
                    url: currentUrl
                }).catch(() => {
                    copyTextToClipboard(currentUrl);
                });
            } else {
                copyTextToClipboard(currentUrl);
            }
        }

        function copyTextToClipboard(text) {
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(text).then(() => {
                    showToast('Đã sao chép liên kết vào bộ nhớ tạm!');
                }).catch(() => fallbackCopy(text));
            } else {
                fallbackCopy(text);
            }
        }

        function fallbackCopy(text) {
            const tempInput = document.createElement('input');
            tempInput.value = text;
            document.body.appendChild(tempInput);
            tempInput.select();
            document.execCommand('copy');
            document.body.removeChild(tempInput);
            showToast('Đã sao chép liên kết vào bộ nhớ tạm!');
        }

        // 4. FLOATING TOAST NOTIFICATION
        function showToast(msg) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toast-message');
            toastMsg.innerText = msg;
            toast.classList.remove('opacity-0', 'translate-y-4', 'pointer-events-none');
            toast.classList.add('opacity-100', 'translate-y-0');

            setTimeout(() => {
                toast.classList.remove('opacity-100', 'translate-y-0');
                toast.classList.add('opacity-0', 'translate-y-4', 'pointer-events-none');
            }, 3000);
        }

        // 5. QR CODE MODAL LOGIC
        const qrModal = document.getElementById('qr-modal');
        const qrModalCard = document.getElementById('qr-modal-card');
        const modalUrlInput = document.getElementById('modal-url-input');

        function openQrModal() {
            const currentUrl = window.location.href;
            modalUrlInput.value = currentUrl;

            const qrBox = document.getElementById('qrcode-box');
            qrBox.innerHTML = '';

            new QRCode(qrBox, {
                text: currentUrl,
                width: 140,
                height: 140,
                colorDark: "#0284C7",
                colorLight: "#FFFFFF",
                correctLevel: QRCode.CorrectLevel.H
            });

            qrModal.classList.remove('hidden');
            setTimeout(() => {
                qrModal.classList.remove('opacity-0');
                qrModalCard.classList.remove('scale-95');
                qrModalCard.classList.add('scale-100');
            }, 10);
        }

        function closeQrModal() {
            qrModal.classList.add('opacity-0');
            qrModalCard.classList.remove('scale-100');
            qrModalCard.classList.add('scale-95');
            setTimeout(() => {
                qrModal.classList.add('hidden');
            }, 300);
        }

        qrModal.addEventListener('click', (e) => {
            if (e.target === qrModal) closeQrModal();
        });

        function copyToClipboardModal() {
            copyTextToClipboard(modalUrlInput.value);
            const modalBtnText = document.getElementById('modal-copy-text');
            modalBtnText.innerText = 'Đã chép!';
            setTimeout(() => {
                modalBtnText.innerText = 'Sao chép';
            }, 2000);
        }

        function downloadQrCode() {
            const img = document.querySelector('#qrcode-box img');
            const canvas = document.querySelector('#qrcode-box canvas');

            let src = '';
            if (img && img.src) {
                src = img.src;
            } else if (canvas) {
                src = canvas.toDataURL('image/png');
            }

            if (src) {
                const a = document.createElement('a');
                a.href = src;
                a.download = 'bio-qrcode.png';
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
            } else {
                showToast('Không thể tạo ảnh QR để tải về!');
            }
        }
    </script>
